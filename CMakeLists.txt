cmake_minimum_required(VERSION 3.20)
project(recomp_audio_core VERSION 0.1.0 LANGUAGES C CXX)
option(RECOMP_AUDIO_BUILD_TESTS "Build independent donor comparison fixtures" ON)
add_library(recomp_audio_core STATIC src/recomp_audio.c src/psx_audio.c)
add_library(recomp-audio::core ALIAS recomp_audio_core)
target_include_directories(recomp_audio_core PUBLIC include)
target_compile_features(recomp_audio_core PUBLIC c_std_99)
if(NOT MSVC)
  target_link_libraries(recomp_audio_core PUBLIC m)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(recomp_audio_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(RECOMP_AUDIO_BUILD_TESTS)
  enable_testing()
  set(PROFILE genesis)
  set(DONOR "${CMAKE_CURRENT_SOURCE_DIR}/tests/oracles/segagenesisrecomp.h")
  configure_file(tests/oracle_impl.c.in oracle_genesis.c @ONLY)
  set(PROFILE nes)
  set(DONOR "${CMAKE_CURRENT_SOURCE_DIR}/tests/oracles/nesrecomp.h")
  configure_file(tests/oracle_impl.c.in oracle_nes.c @ONLY)
  add_executable(audio_differential tests/audio_test.c tests/oracle_dispatch.c
    "${CMAKE_CURRENT_BINARY_DIR}/oracle_genesis.c" "${CMAKE_CURRENT_BINARY_DIR}/oracle_nes.c")
  target_include_directories(audio_differential PRIVATE tests)
  target_link_libraries(audio_differential PRIVATE recomp-audio::core)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(audio_differential PRIVATE -ffp-contract=off)
  endif()
  add_test(NAME frozen_donor_differential COMMAND audio_differential)
  add_executable(cpp_api tests/cpp_api.cpp)
  target_link_libraries(cpp_api PRIVATE recomp-audio::core)
  add_test(NAME cpp_api COMMAND cpp_api)
endif()

if(RECOMP_AUDIO_BUILD_TESTS)
  add_executable(native_queue tests/native_queue.cpp)
  target_link_libraries(native_queue PRIVATE recomp-audio::core)
  add_test(NAME native_queue COMMAND native_queue)
endif()
